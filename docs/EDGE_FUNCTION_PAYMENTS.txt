================================================================================
CÓDIGO DE LA SUPABASE EDGE FUNCTION DE PAGOS EN LÍNEA (payment-service)
================================================================================
Ubicación en Supabase CLI: supabase/functions/payment-service/index.ts

PASOS DE INSTALACIÓN Y DESPLIEGUE EN SUPABASE:
--------------------------------------------------------------------------------
1. Instala Supabase CLI en tu terminal:
   npm install -g supabase

2. Crea la función en tu proyecto:
   supabase functions new payment-service

3. Copia el código TypeScript de abajo en: supabase/functions/payment-service/index.ts

4. Configura tus Secrets en Supabase Dashboard (Settings > Secrets):
   - STRIPE_CLIENT_ID (Client ID de tu Stripe Connect App)
   - STRIPE_SECRET_KEY (Secret Key de tu plataforma Tragalero)
   - MERCADOPAGO_CLIENT_ID (Client ID de tu App en Mercado Pago Developers)
   - MERCADOPAGO_CLIENT_SECRET (Client Secret de tu App en Mercado Pago Developers)
   - SUPABASE_URL (URL de tu proyecto Supabase)
   - SUPABASE_SERVICE_ROLE_KEY (Service Role Key para guardar configuraciones de pago)

5. Despliega la función:
   supabase functions deploy payment-service --no-verify-jwt

================================================================================
CÓDIGO TYPESCRIPT (index.ts)
================================================================================

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2.38.4";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

serve(async (req) => {
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: corsHeaders });
  }

  try {
    const supabaseUrl = Deno.env.get("SUPABASE_URL") ?? "";
    const supabaseServiceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ?? "";
    const supabase = createClient(supabaseUrl, supabaseServiceKey);

    const body = await req.json();
    const { action, user_id, order_id, code, redirect_uri, amount, currency, provider } = body;

    // 1. GENERAR URLS DE REDIRECCIÓN OAUTH (Para conectar cuentas de restaurantes)
    if (action === "get_stripe_oauth_url") {
      const stripeClientId = Deno.env.get("STRIPE_CLIENT_ID");
      if (!stripeClientId) {
        return new Response(JSON.stringify({ error: "STRIPE_CLIENT_ID no configurado" }), { status: 400, headers: corsHeaders });
      }
      const url = `https://connect.stripe.com/oauth/authorize?response_type=code&client_id=${stripeClientId}&scope=read_write&state=${user_id}&redirect_uri=${encodeURIComponent(redirect_uri)}`;
      return new Response(JSON.stringify({ url }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
    }

    if (action === "get_mp_oauth_url") {
      const mpClientId = Deno.env.get("MERCADOPAGO_CLIENT_ID");
      if (!mpClientId) {
        return new Response(JSON.stringify({ error: "MERCADOPAGO_CLIENT_ID no configurado" }), { status: 400, headers: corsHeaders });
      }
      const url = `https://auth.mercadopago.com.mx/authorization?client_id=${mpClientId}&response_type=code&platform_id=mp&state=${user_id}&redirect_uri=${encodeURIComponent(redirect_uri)}`;
      return new Response(JSON.stringify({ url }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
    }

    // 2. INTERCAMBIO DE CÓDIGO OAUTH POR TOKENS/ACCOUNT_ID (OAuth Callback)
    if (action === "oauth_callback" && code && user_id) {
      // Intercambio con Stripe
      const stripeSecretKey = Deno.env.get("STRIPE_SECRET_KEY");
      if (stripeSecretKey) {
        const stripeRes = await fetch("https://connect.stripe.com/oauth/token", {
          method: "POST",
          headers: { "Content-Type": "application/x-www-form-urlencoded" },
          body: new URLSearchParams({
            client_secret: stripeSecretKey,
            code: code,
            grant_type: "authorization_code"
          })
        });
        const stripeData = await stripeRes.json();
        if (stripeData.stripe_user_id) {
          await supabase.from("tragalero_payment_config").upsert({
            user_id: user_id,
            stripe_connected: true,
            stripe_enabled: true,
            stripe_account_id: stripeData.stripe_user_id,
            updated_at: new Date().toISOString()
          }, { onConflict: "user_id" });
          return new Response(JSON.stringify({ success: true, provider: "stripe", account_id: stripeData.stripe_user_id }), { headers: corsHeaders });
        }
      }

      // Intercambio con Mercado Pago
      const mpClientSecret = Deno.env.get("MERCADOPAGO_CLIENT_SECRET");
      const mpClientId = Deno.env.get("MERCADOPAGO_CLIENT_ID");
      if (mpClientSecret && mpClientId) {
        const mpRes = await fetch("https://api.mercadopago.com/oauth/token", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            client_secret: mpClientSecret,
            client_id: mpClientId,
            grant_type: "authorization_code",
            code: code,
            redirect_uri: redirect_uri
          })
        });
        const mpData = await mpRes.json();
        if (mpData.user_id && mpData.access_token) {
          await supabase.from("tragalero_payment_config").upsert({
            user_id: user_id,
            mercadopago_connected: true,
            mercadopago_enabled: true,
            mercadopago_user_id: mpData.user_id.toString(),
            mercadopago_access_token: mpData.access_token,
            mercadopago_refresh_token: mpData.refresh_token,
            updated_at: new Date().toISOString()
          }, { onConflict: "user_id" });
          return new Response(JSON.stringify({ success: true, provider: "mercadopago", user_id: mpData.user_id }), { headers: corsHeaders });
        }
      }
    }

    // 3. CREACIÓN DE SESIÓN DE PAGO (Checkout Diner)
    if (action === "create_checkout_session") {
      const { data: payConfig } = await supabase
        .from("tragalero_payment_config")
        .select("*")
        .eq("user_id", user_id)
        .single();

      if (!payConfig) {
        return new Response(JSON.stringify({ error: "Sin configuración de pagos." }), { status: 400, headers: corsHeaders });
      }

      if (provider === "stripe" && payConfig.stripe_account_id) {
        const stripeSecretKey = Deno.env.get("STRIPE_SECRET_KEY");
        const checkoutParams = new URLSearchParams({
          "payment_method_types[]": "card",
          "line_items[0][price_data][currency]": (currency || "mxn").toLowerCase(),
          "line_items[0][price_data][product_data][name]": `Pedido #${order_id}`,
          "line_items[0][price_data][unit_amount]": Math.round(amount * 100).toString(),
          "line_items[0][quantity]": "1",
          "mode": "payment",
          "success_url": `${redirect_uri}?payment=success&order_id=${order_id}`,
          "cancel_url": `${redirect_uri}?payment=cancel&order_id=${order_id}`,
          "client_reference_id": order_id.toString()
        });

        const stripeRes = await fetch("https://api.stripe.com/v1/checkout/sessions", {
          method: "POST",
          headers: {
            "Authorization": `Bearer ${stripeSecretKey}`,
            "Stripe-Account": payConfig.stripe_account_id,
            "Content-Type": "application/x-www-form-urlencoded"
          },
          body: checkoutParams
        });
        const stripeData = await stripeRes.json();
        return new Response(JSON.stringify({ checkout_url: stripeData.url }), { headers: corsHeaders });
      }

      if (provider === "mercadopago" && payConfig.mercadopago_access_token) {
        const preferenceBody = {
          items: [{ title: `Pedido Tragalero #${order_id}`, unit_price: Number(amount), quantity: 1, currency_id: "MXN" }],
          external_reference: order_id.toString(),
          back_urls: {
            success: `${redirect_uri}?payment=success&order_id=${order_id}`,
            failure: `${redirect_uri}?payment=failure&order_id=${order_id}`
          },
          auto_return: "approved"
        };
        const mpRes = await fetch("https://api.mercadopago.com/checkout/preferences", {
          method: "POST",
          headers: {
            "Authorization": `Bearer ${payConfig.mercadopago_access_token}`,
            "Content-Type": "application/json"
          },
          body: JSON.stringify(preferenceBody)
        });
        const mpData = await mpRes.json();
        return new Response(JSON.stringify({ checkout_url: mpData.init_point || mpData.sandbox_init_point }), { headers: corsHeaders });
      }
    }

    // 4. WEBHOOKS DE PAGO
    if (action === "webhook_stripe") {
      const event = body;
      if (event.type === "checkout.session.completed") {
        const session = event.data.object;
        await supabase.from("tragalero_orders")
          .update({ payment_status: "APROBADO", payment_id: session.id, payment_provider: "stripe" })
          .eq("id", session.client_reference_id);
      }
      return new Response(JSON.stringify({ received: true }), { headers: corsHeaders });
    }

    return new Response(JSON.stringify({ error: "Acción no válida" }), { status: 400, headers: corsHeaders });

  } catch (err) {
    return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: corsHeaders });
  }
});
